# The app and the site are one page, so there is nothing to steer a crawler
# away from: the copy that describes the product is in the same document as
# the tool itself.
# /script.js and /api/send are the visitor-statistics script and the
# endpoint it reports to; neither carries page content, so every group
# below keeps crawlers off those two addresses and nothing else.

Disallow: /api/send
Disallow: /script.js
Allow: /
